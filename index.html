  var SERVICES = {

    'quick-trot': { name: 'Quick Trot', duration: 20, price: 5 },

  // Full postcodes belonging to the Holland Park neighborhood in Diemen.

  var ALLOWED_POSTCODES = [

    '1112WT', '1112WW', '1112WX', '1112WZ',

    '1112ZA', '1112ZB', '1112ZC', '1112ZD', '1112ZE',

    '1112ZG', '1112ZH', '1112ZJ', '1112ZK', '1112ZL'

  ];


  function normalizePostcode(value){

    return value.toUpperCase().replace(/\s+/g, '');

  }


  function formatPostcode(value){

    var normalized = normalizePostcode(value);

    return normalized.length === 6 ? normalized.slice(0, 4) + ' ' + normalized.slice(4) : value.trim().toUpperCase();

  }


  function isValidDutchPostcode(value){

    return /^[1-9][0-9]{3}[A-Z]{2}$/.test(normalizePostcode(value));

  }


  function isAllowedPostcode(value){

    return ALLOWED_POSTCODES.indexOf(normalizePostcode(value)) !== -1;

  }


  function to12Hour(hours, minutes){

    var period = hours >= 12 ? 'PM' : 'AM';

    var h = hours % 12;

    if (h === 0) h = 12;

    var m = minutes < 10 ? '0' + minutes : minutes;

    return h + ':' + m + ' ' + period;

  }


  function formatDateLong(dateStr){

    var parts = dateStr.split('-');

    var d = new Date(parseInt(parts[0]), parseInt(parts[1]) - 1, parseInt(parts[2]));

    return d.toLocaleDateString(undefined, { weekday:'long', month:'long', day:'numeric' });

  }


  var memoryBookings = {}; // fallback used only if platform storage is unavailable

  var storageAvailable = (typeof window.storage !== 'undefined' && window.storage !== null);


  async function getBookingsForDate(dateStr){

    if (storageAvailable){

      try {

        var result = await window.storage.get('bookings:' + dateStr, true);

        return result ? JSON.parse(result.value) : [];

      } catch (err) {

        // no bookings saved yet for this date

        return [];

      }

    }

    return memoryBookings[dateStr] ? memoryBookings[dateStr].slice() : [];

  }


  async function saveBooking(dateStr, booking){

    var existing = await getBookingsForDate(dateStr);

    existing.push(booking);


    if (storageAvailable){

      try {

        var result = await window.storage.set('bookings:' + dateStr, JSON.stringify(existing), true);

        return !!result;

      } catch (err) {

        console.error('Platform storage failed, falling back to in-memory:', err);

        storageAvailable = false; // stop retrying platform storage this session

      }

    }

    memoryBookings[dateStr] = existing;

    return true;

  }


  function overlaps(aStart, aEnd, bStart, bEnd){

    return aStart < bEnd && aEnd > bStart;

  }


  document.getElementById('bookingForm').addEventListener('submit', async function(e){

    e.preventDefault();


    var errorBox = document.getElementById('slotError');

    var submitBtn = document.getElementById('submitBtn');

    errorBox.classList.remove('show');


    var ownerName = document.getElementById('ownerName').value.trim();

    var dogName = document.getElementById('dogName').value.trim();

    var serviceId = document.getElementById('serviceChoice').value;

    var service = SERVICES[serviceId];

    var pickupPostcode = document.getElementById('pickupPostcode').value.trim();

    var dateStr = document.getElementById('date').value;

    var timeStr = document.getElementById('time').value;


    if (!ownerName || !dogName || !service || !pickupPostcode || !dateStr || !timeStr){

      errorBox.textContent = 'Please fill in your name, your dog\'s name, the pickup postcode, the date, and the time before requesting a walk.';

      errorBox.classList.add('show');

      return;

    }


    if (!isValidDutchPostcode(pickupPostcode)){

      errorBox.textContent = 'Enter a valid Dutch postcode, for example 1112 AB.';

      errorBox.classList.add('show');

      return;

    }


    if (!isAllowedPostcode(pickupPostcode)){

      errorBox.textContent = 'That pickup postcode is outside Bavly\'s Holland Park walking area.';

      errorBox.classList.add('show');

      return;

    }


    pickupPostcode = formatPostcode(pickupPostcode);

    document.getElementById('pickupPostcode').value = pickupPostcode;

    verifiedPostcode = pickupPostcode;


    var duration = service.duration;


    var timeParts = timeStr.split(':');

    var startMinutes = parseInt(timeParts[0]) * 60 + parseInt(timeParts[1]);

    var endMinutes = startMinutes + duration;


    submitBtn.disabled = true;

    submitBtn.textContent = 'Checking availability…';


    var existingBookings = await getBookingsForDate(dateStr);

    var conflict = existingBookings.some(function(b){

      return overlaps(startMinutes, endMinutes, b.start, b.end);

    });


    if (conflict){

      submitBtn.disabled = false;

      submitBtn.textContent = 'Request this walk';

      errorBox.textContent = 'This time is taken — please pick another time.';

      errorBox.classList.add('show');

      return;

    }


    var saved = await saveBooking(dateStr, {

      start: startMinutes,

      end: endMinutes,

      dogName: dogName,

      duration: duration,

      serviceId: serviceId,

      price: service.price,

      pickupPostcode: pickupPostcode

    });

    if (!saved){

      errorBox.textContent = "Something went wrong saving your booking — please try again.";

      errorBox.classList.add('show');

      return;

    var pickupH = Math.floor(startMinutes / 60);

    document.getElementById('detailPrice').textContent = '€' + service.price;

    document.getElementById('detailPostcode').textContent = pickupPostcode;

  function isPastDate(y, m, d){

  function toISODate(y, m, d){

    var mm = (m + 1) < 10 ? '0' + (m + 1) : (m + 1);

    var dd = d < 10 ? '0' + d : d;

    return y + '-' + mm + '-' + dd;

  }


  function renderCalendar(){

    var m = calViewDate.getMonth();

    document.getElementById('calMonthLabel').textContent =

      calViewDate.toLocaleDateString(undefined, { month: 'long', year: 'numeric' });

    var firstWeekday = new Date(y, m, 1).getDay();

    var daysInMonth = new Date(y, m + 1, 0).getDate();


    for (var i = 0; i < firstWeekday; i++){

      var empty = document.createElement('div');

      empty.className = 'cal-day empty';

      grid.appendChild(empty);

    }


    for (var d = 1; d <= daysInMonth; d++){

      cell.className = 'cal-day';

      cell.textContent = d;

      var weekday = new Date(y, m, d).getDay();

      var iso = toISODate(y, m, d);


      if (isPastDate(y, m, d)){

        cell.classList.add('past');

      } else if (AVAILABLE_WEEKDAYS.indexOf(weekday) === -1){

        cell.classList.add('unavailable');

  function generateTimeSlots(){

    var timeSelect = document.getElementById('time');

    timeSelect.innerHTML = '';


    if (!selectedDateStr){

      timeSelect.disabled = true;

      timeSelect.innerHTML = '<option value="">Pick a date first</option>';

      return;

    }


    var selectedService = SERVICES[document.getElementById('serviceChoice').value];

    var duration = selectedService ? selectedService.duration : 0;

    var lastStart = DAY_END_MIN - duration;


    if (lastStart < DAY_START_MIN){

      timeSelect.disabled = true;

      timeSelect.innerHTML = '<option value="">No slots long enough that day</option>';

      return;

    }


    timeSelect.disabled = false;

    var options = '<option value="">Choose a time</option>';

    for (var mins = DAY_START_MIN; mins <= lastStart; mins += 15){

      var h = Math.floor(mins / 60);

      var mm = mins % 60;

      var value = (h < 10 ? '0' + h : h) + ':' + (mm < 10 ? '0' + mm : mm);

      options += '<option value="' + value + '">' + to12Hour(h, mm) + '</option>';

    }

    timeSelect.innerHTML = options;

  }


  document.getElementById('serviceChoice').addEventListener('change', function(){

    generateTimeSlots();

  });


  renderCalendar();
